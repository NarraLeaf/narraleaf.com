---
title: "LayeredDefinition"
description: 分层 `Image` 的 `src` 结构：从下到上的图层堆叠，以及每组互斥变体的默认标签
---
层叠图像的 src，参见 [Image](/zh/docs/narraleaf-react/core/elements/image#%E5%9B%BE%E5%B1%82%E6%A8%A1%E5%BC%8Flayered)

```typescript
export type LayeredDefinition<L extends LayerGroupDefinition = LayerGroupDefinition> = {
    /**
     * 图层栈，从底到顶。数组顺序即叠放顺序
     */
    layers: L;
    /**
     * 每个组一个标签，顺序任意。提供同一标签集合的层共享同一个组，
     * 因此也共享这一个默认值
     */
    defaults: readonly LayerTagsOf<L>[];
};
```

```typescript
/**
 * 层叠图像的一层，从底到顶。可以是常量 src、`null`、变体映射，或解析函数
 */
export type LayerSlot = string | null | LayerVariants | LayerResolver;
export type LayerGroupDefinition = readonly LayerSlot[];

/**
 * 一层的互斥变体集合，以标签为键
 *
 * `null` 表示该标签下这一层不绘制
 *
 * 标签集合就是组的身份：提供**相同**标签集合的每一层都由同一个组驱动，
 * 这正是一次 `char(["angry"])` 能同时移动眉毛、眼睛和嘴巴的原因。
 * 要跟随一个组，就在该层上重复它的完整标签集合，并在该层不绘制的标签上写 `null`。
 * 只提供集合的一部分，等于声明了**另一个**组，共享的标签随即冲突
 */
export type LayerVariants = Record<string, string | null>;

/**
 * 根据当前生效的标签派生该层的 src。自身不声明任何标签
 *
 * 解析函数是不透明的，它可能返回的 src 对预载器不可见，会在首次使用时才被获取。
 * 请优先使用 LayerVariants，包括用于跟随其他层的组的层（它已经不再需要解析函数），
 * 只把解析函数留给确实无法枚举的来源
 */
export type LayerResolver = (tags: ReadonlySet<string>) => string | null;

/**
 * 图层栈所声明的全部标签的联合类型
 */
export type LayerTagsOf<L> = L extends readonly (infer E)[]
    ? (E extends LayerResolver ? never : E extends LayerVariants ? keyof E & string : never)
    : never;
```

标签会从定义中自动推导，因此无需显式传入类型参数：

```typescript
const yuko = new Image({
    src: {
        layers: [
            "yuko/body.png",
            {uniform: "yuko/uniform.png", casual: "yuko/casual.png"},
            {happy: "yuko/happy.png", sad: "yuko/sad.png"},
        ],
        defaults: ["uniform", "happy"],
    },
});

yuko.char(["sad"]);  // 正确
yuko.char(["sda"]);  // 编译期报错
```
